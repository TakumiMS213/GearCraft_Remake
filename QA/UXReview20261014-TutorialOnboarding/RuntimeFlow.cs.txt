 var hits=new System.Collections.Generic.List<UnityEngine.EventSystems.RaycastResult>();events.RaycastAll(data,hits);check(hits.Count>0,"raycast hits "+o.name);data.pointerCurrentRaycast=hits[0];data.pointerPressRaycast=hits[0];return data;
};
System.Action<UnityEngine.UI.Button> submitNow=b=>{
 check(b!=null && b.IsActive() && b.IsInteractable(),"submit available "+(b==null?"null":b.name));
 TooltipSystem.FindInstance().Hide();var p=pointer(b.gameObject);
 check(UnityEngine.EventSystems.ExecuteEvents.GetEventHandler<UnityEngine.EventSystems.ISubmitHandler>(p.pointerCurrentRaycast.gameObject)==b.gameObject,"top raycast reaches button "+b.name);
 b.Select();UnityEngine.EventSystems.ExecuteEvents.Execute(b.gameObject,new UnityEngine.EventSystems.BaseEventData(events),UnityEngine.EventSystems.ExecuteEvents.submitHandler);
};
System.Action<UnityEngine.UI.Button> submit=b=>{pendingAction=()=>submitNow(b);};
System.Action<string> inspect=label=>{
 Canvas.ForceUpdateCanvases();var panel=(RectTransform)field("messagePanel");var root=(RectTransform)panel.GetComponentInParent<Canvas>().transform;
 var bounds=RectTransformUtility.CalculateRelativeRectTransformBounds(root,panel);
 check(bounds.min.x>=root.rect.xMin-.5f && bounds.max.x<=root.rect.xMax+.5f && bounds.min.y>=root.rect.yMin-.5f && bounds.max.y<=root.rect.yMax+.5f,label+" panel within canvas");
 var rects=new System.Collections.Generic.List<Rect>();
 foreach(var t in panel.GetComponentsInChildren<TMPro.TMP_Text>()){t.ForceMeshUpdate();check(!t.isTextOverflowing && !t.isTextTruncated,label+" text fits "+t.name+" "+t.text.Replace("\n"," / "));}
 foreach(var b in panel.GetComponentsInChildren<UnityEngine.UI.Button>()){var r=RectTransformUtility.CalculateRelativeRectTransformBounds(panel,b.transform);var rr=new Rect(r.min.x,r.min.y,r.size.x,r.size.y);check(rr.xMin>=panel.rect.xMin-.5f && rr.xMax<=panel.rect.xMax+.5f,label+" button within panel "+b.name);foreach(var prior in rects)check(!prior.Overlaps(rr),label+" buttons do not overlap");rects.Add(rr);}
 log("SCREEN "+Screen.width+"x"+Screen.height+" "+label+" action="+text("taskActionText").text);
 ScreenCapture.CaptureScreenshot(folder+"runtime-"+label+".png");
};
System.Action<UpgradePartSO> beginNow=part=>{
 draggingPart=part;source=find("ShopItem_"+part.name);check(source!=null,"shop row exists "+part.name);
 var scroll=shop.shopParent.GetComponentInParent<UnityEngine.UI.ScrollRect>();if(scroll!=null){scroll.StopMovement();scroll.verticalNormalizedPosition=1f;Canvas.ForceUpdateCanvases();}
 drag=pointer(source);check(UnityEngine.EventSystems.ExecuteEvents.GetEventHandler<UnityEngine.EventSystems.IBeginDragHandler>(drag.pointerCurrentRaycast.gameObject)==source,"row drag handler receives raycast");drag.pointerDrag=source;drag.dragging=true;
 UnityEngine.EventSystems.ExecuteEvents.Execute(source,drag,UnityEngine.EventSystems.ExecuteEvents.beginDragHandler);
 check(gridUI.IsShopDragging,"real begin-drag starts preview");
 int px=-1,py=-1;for(int y=0;y<grid.GridSize && px<0;y++)for(int x=0;x<grid.GridSize;x++)if(grid.CanPlace(part,x,y,0)){px=x;py=y;break;}
 check(px>=0,"valid placement exists");var shape=part.GetRotatedShape(0);int minX=999,minY=999,maxX=-1,maxY=-1;
 for(int y=0;y<shape.GetLength(0);y++)for(int x=0;x<shape.GetLength(1);x++)if(shape[y,x]){minX=Mathf.Min(minX,x);maxX=Mathf.Max(maxX,x);minY=Mathf.Min(minY,y);maxY=Mathf.Max(maxY,y);}
 float size=(float)typeof(UpgradeGridUI).GetField("currentCellSize",flags).GetValue(gridUI);float step=size+gridUI.cellSpacing;float total=grid.GridSize*step-gridUI.cellSpacing;
 Vector3 local=new Vector3(-total/2f+size/2f+(px+(minX+maxX)/2f)*step,total/2f-size/2f-(py+(minY+maxY)/2f)*step,0);
 var canvas=gridUI.GetComponentInParent<Canvas>().rootCanvas;target=RectTransformUtility.WorldToScreenPoint(canvas.renderMode==RenderMode.ScreenSpaceOverlay?null:canvas.worldCamera,gridUI.gridParent.TransformPoint(local));
};
System.Action<UpgradePartSO> begin=part=>{pendingAction=()=>beginNow(part);};
System.Action dropNow=()=>{drag.position=target;UnityEngine.EventSystems.ExecuteEvents.Execute(source,drag,UnityEngine.EventSystems.ExecuteEvents.dragHandler);UnityEngine.EventSystems.ExecuteEvents.Execute(source,drag,UnityEngine.EventSystems.ExecuteEvents.endDragHandler);check(!gridUI.IsShopDragging,"end drag clears preview");};
System.Action drop=()=>{pendingAction=dropNow;};
System.Action<bool,string> finish=(ok,message)=>{
 if(finished)return;finished=true;UnityEditor.EditorApplication.update-=tick;Application.logMessageReceived-=onLog;craft.RecipeCrafted-=onCraft;
 if(gridUI!=null)gridUI.CancelShopDrag();UnityEditor.SessionState.SetBool(prefix+"running",false);UnityEditor.SessionState.SetBool(prefix+"passed",ok);UnityEditor.SessionState.SetInt(prefix+"checks",checks);UnityEditor.SessionState.SetString(prefix+"result",message);
 log((ok?"COMPLETE ":"FAILED ")+message+" checks="+checks+" errors="+errors);
};
onLog=(message,stack,type)=>{if(type==LogType.Error||type==LogType.Exception||type==LogType.Assert){errors++;log("ERROR "+message+"\n"+stack);}};
tick=()=>{try{
 if(!UnityEditor.EditorApplication.isPlaying||UnityEditor.EditorApplication.isPaused)throw new System.Exception("Play stopped/paused");
 double now=UnityEditor.EditorApplication.timeSinceStartup;if(now-started>100)throw new System.Exception("100s watchdog");if(errors>0)throw new System.Exception("Runtime error");
 if(now<next||lastFrame==Time.frameCount)return;next=now+.3;lastFrame=Time.frameCount;UnityEditor.SessionState.SetInt(prefix+"phase",phase);
 if(pendingAction!=null){var action=pendingAction;pendingAction=null;action();return;}
 switch(phase){
 case 0:
  c.OnCraftSceneLeaving();status.CraftTutorials.Reset();status.module_scrap=false;status.punkDrive=false;status.craftWeaponDamagebuff=0;materials.ClearAllMaterials();
  PlayerPrefs.SetInt(CraftTutorialMetrics.CraftMasteryKey,0);PlayerPrefs.SetInt(CraftTutorialMetrics.WeaponMasteryKey,0);
  var auto=(bool[])field("autoStarted");auto[0]=auto[1]=false;craft.TrySelectRecipe(craft.FindUnlockedRecipe("GearCraft"));tabs.SwitchTab(0);phase++;break;
 case 1:
  if(!active())return;check(status.CraftTutorials.craft.stepIndex==0,"first task starts automatically");check(button("hintButton").IsActive(),"hint available immediately");check(!button("affordableButton").gameObject.activeSelf && !button("shortageButton").gameObject.activeSelf,"assessment answers hidden before recipe selection");check(frameCount()==1,"initial guide highlights recipe region");inspect("01-craft-start");phase++;break;
 case 2:submit(find("Recipe_ScrapModule").GetComponent<UnityEngine.UI.Button>());phase++;break;
 case 3:
  check(button("affordableButton").IsActive() && button("shortageButton").IsActive(),"assessment answers appear after selecting ScrapModule");check(frameCount()==2,"guide now highlights two materials");inspect("02-material-assessment");submit(button("hintButton"));phase++;break;
 case 4:check(status.CraftTutorials.craft.hintLevel==1,"first hint is conceptual");submit(button("hintButton"));phase++;break;
 case 5:check(status.CraftTutorials.craft.hintLevel==2,"second hint shows reading area");submit(button("hintButton"));phase++;break;
 case 6:
  check(status.CraftTutorials.craft.hintLevel==3,"third hint gives concrete operation");inspect("03-hint-three");submit(button("practiceButton"));phase++;break;
 case 7:
  check(!text("bodyText").gameObject.activeSelf && text("taskActionText").gameObject.activeSelf,"folding hides explanation and keeps current operation");inspect("03b-materials-folded");submit(button("hintButton"));phase++;break;
 case 8:
  check(text("bodyText").gameObject.activeSelf && status.CraftTutorials.craft.hintRequests==3,"last hint reopens without duplicate use count");submit(button("shortageButton"));phase++;break;

  check(status.CraftTutorials.craft.showingFeedback && CraftManager.HasEnoughMaterials(craft.SelectedRecipe),"shortage judgment grants required materials and shows feedback");inspect("04-materials-supplied");submit(button("nextButton"));phase++;break;
 case 10:
  check(status.CraftTutorials.craft.stepIndex==1 && frameCount()==1,"next task guides actual creation");inspect("05-craft-action");submit(craft.craftButton);phase++;break;
 case 11:
  check(craft.IsCrafting,"craft presentation is running");check(((CanvasGroup)field("messageGroup")).alpha==0 && frameCount()==0,"tutorial text and frames hidden during craft");var statusText=(TMPro.TMP_Text)typeof(CraftManager).GetField("craftStatusText",flags).GetValue(craft);check(!statusText.gameObject.activeSelf,"no crafting status during animation");check(craftEvents==0,"completion event waits for presentation");int gear=materials.gear,core=materials.moduleCore_lv1;craft.OnCraftButton();check(gear==materials.gear && core==materials.moduleCore_lv1,"duplicate craft input does not consume again");ScreenCapture.CaptureScreenshot(folder+"runtime-06-craft-animation.png");phase++;break;
 case 12:
  if(craft.IsCrafting)return;check(craftEvents==1 && status.CraftTutorials.craft.showingFeedback,"one completion notification after animation");check(((TMPro.TMP_Text)typeof(CraftManager).GetField("craftStatusText",flags).GetValue(craft)).text=="作成済み","crafted status shown after final fade");inspect("07-craft-complete");submit(button("nextButton"));phase++;break;
 case 13:

  if(!active() || (int)field("currentTab")!=1)return;grid=UpgradeGridManager.Instance;gridUI=grid.gridUI;shop=UnityEngine.Object.FindFirstObjectByType<UpgradeShopManager>();check(grid.PlacedParts.Count==0,"fresh weapon grid empty");check(frameCount()==1 && button("hintButton").IsActive(),"initial upgrade highlights shop and provides hint");inspect("10-upgrade-start");source=find("ShopItem_Damage+1");check(source!=null,"funded practice row visible");drag=pointer(source);pendingAction=()=>UnityEngine.EventSystems.ExecuteEvents.Execute(source,drag,UnityEngine.EventSystems.ExecuteEvents.pointerEnterHandler);phase++;break;

 }
}catch(System.Exception ex){finish(false,ex.ToString());}};
