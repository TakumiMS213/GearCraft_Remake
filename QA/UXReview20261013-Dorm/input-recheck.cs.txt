System.Func<Transform,string> pathOf=null;pathOf=delegate(Transform t){return t.parent==null?t.name:pathOf(t.parent)+"/"+t.name;};
System.Func<string,GameObject> find=delegate(string path){foreach(var tr in UnityEngine.Object.FindObjectsByType<Transform>(FindObjectsInactive.Include,FindObjectsSortMode.None))if(tr.gameObject.scene.IsValid()&&(pathOf(tr)==path||tr.name==path))return tr.gameObject;throw new System.Exception("UI object not found: "+path);};
System.Func<GameObject,Vector2> center=delegate(GameObject go){var rt=go.GetComponent<RectTransform>();var c=go.GetComponentInParent<Canvas>();return RectTransformUtility.WorldToScreenPoint(c!=null&&c.renderMode!=RenderMode.ScreenSpaceOverlay?c.worldCamera:null,rt.TransformPoint(rt.rect.center));};
System.Func<Vector2,UnityEngine.EventSystems.PointerEventData> pointer=delegate(Vector2 pos){var es=UnityEngine.EventSystems.EventSystem.current;var ev=new UnityEngine.EventSystems.PointerEventData(es){position=pos,button=UnityEngine.EventSystems.PointerEventData.InputButton.Left};var hits=new System.Collections.Generic.List<UnityEngine.EventSystems.RaycastResult>();es.RaycastAll(ev,hits);if(hits.Count==0)throw new System.Exception("No raycast at "+pos);ev.pointerCurrentRaycast=hits[0];ev.pointerPressRaycast=hits[0];return ev;};
System.Action<string> click=delegate(string path){var go=find(path);var ev=pointer(center(go));foreach(var entrance in UnityEngine.Object.FindObjectsByType<GearCraft.Gallery.GalleryEntranceButton>(FindObjectsSortMode.None))UnityEngine.EventSystems.ExecuteEvents.Execute(entrance.gameObject,ev,UnityEngine.EventSystems.ExecuteEvents.pointerExitHandler);var handler=UnityEngine.EventSystems.ExecuteEvents.GetEventHandler<UnityEngine.EventSystems.IPointerClickHandler>(ev.pointerCurrentRaycast.gameObject);if(handler!=go&&!(handler!=null&&handler.transform.IsChildOf(go.transform)))throw new System.Exception("Blocked "+path+" by "+pathOf(ev.pointerCurrentRaycast.gameObject.transform));UnityEngine.EventSystems.ExecuteEvents.Execute(handler,ev,UnityEngine.EventSystems.ExecuteEvents.pointerDownHandler);UnityEngine.EventSystems.ExecuteEvents.Execute(handler,ev,UnityEngine.EventSystems.ExecuteEvents.pointerUpHandler);UnityEngine.EventSystems.ExecuteEvents.Execute(handler,ev,UnityEngine.EventSystems.ExecuteEvents.pointerClickHandler);System.IO.File.AppendAllText("QA/UXReview20261013-Dorm/actions.txt","CLICK "+path+" @ "+ev.position+"\n");};
