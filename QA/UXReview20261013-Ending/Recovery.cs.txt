using UnityEditor;
using UnityEngine;
using System;
using System.IO;

[InitializeOnLoad]
internal static class UXEndingRestoreOnce
{
    private const string Log = "QA/UXReview20261013-Ending/recovery.txt";
    private static readonly double Started = EditorApplication.timeSinceStartup;
    static UXEndingRestoreOnce()
    {
        if (!SessionState.GetBool("UXEnding20261013.active", false)) return;
        EditorApplication.update += Tick;
    }
    private static async void Tick()
    {
        if (EditorApplication.timeSinceStartup - Started > 60)
        {
            EditorApplication.update -= Tick;
            File.AppendAllText(Log, DateTime.Now.ToString("s") + " recovery timeout before restore\n");
            return;
        }
        if (EditorApplication.isCompiling || EditorApplication.isUpdating) return;
        if (EditorApplication.isPlaying)
        {
            EditorApplication.isPlaying = false;
            return;
        }
        if (EditorApplication.isPlayingOrWillChangePlaymode) return;
        EditorApplication.update -= Tick;
        try
        {
            string restored = Restore();
            string verified = Verify();
            File.AppendAllText(Log, DateTime.Now.ToString("s") + " " + restored + "\n" + verified + "\n");
            bool connected = await MCPForUnity.Editor.Services.MCPServiceLocator.Bridge.StartAsync();
            File.AppendAllText(Log, DateTime.Now.ToString("s") + " reconnect=" + connected + "\n");
        }
        catch (Exception exception)
        {
            File.AppendAllText(Log, DateTime.Now.ToString("s") + " recovery failed: " + exception + "\n");
        }
    }
    private static string Restore()
    {

string[] testLines=System.IO.File.ReadAllLines("/tmp/gearcraft-uxending-playerprefs-backup.txt");
var testExpected=new System.Collections.Generic.HashSet<string>(new[]{"GearCraft.Gallery.Progress.v1","GearCraft.CraftTutorial.Metrics.Craft.v5","GearCraft.CraftTutorial.Metrics.WeaponCustom.v5","GearCraft.Main.OpeningNotice.Seen.v1","GearCraft.CraftTutorial.Mastery.Craft.v5","GearCraft.CraftTutorial.Mastery.WeaponCustom.v5","GearCraft.CraftTutorial.AutoPlayed.Craft.v4","GearCraft.CraftTutorial.AutoPlayed.WeaponCustom.v3"});
var testRecords=new System.Collections.Generic.List<string[]>();
foreach(var testLine in testLines) {
 var testRow=testLine.Split('|');
 if(testRow.Length!=4 || !testExpected.Remove(testRow[0])) throw new System.Exception("Invalid backup key");
 bool.Parse(testRow[1]);
 if(testRow[2]=="string") System.Convert.FromBase64String(testRow[3]); else if(testRow[2]=="int") int.Parse(testRow[3]); else throw new System.Exception("Invalid backup type");
 testRecords.Add(testRow);
}
if(testExpected.Count!=0) throw new System.Exception("Incomplete backup");
foreach(var testRow in testRecords) {
 if(!bool.Parse(testRow[1])) PlayerPrefs.DeleteKey(testRow[0]);
 else if(testRow[2]=="string") PlayerPrefs.SetString(testRow[0],System.Text.Encoding.UTF8.GetString(System.Convert.FromBase64String(testRow[3])));
 else PlayerPrefs.SetInt(testRow[0],int.Parse(testRow[3]));
}
PlayerPrefs.Save();
foreach(var testRow in testRecords) {
 bool testExists=bool.Parse(testRow[1]);
 if(PlayerPrefs.HasKey(testRow[0])!=testExists) throw new System.Exception("Presence mismatch "+testRow[0]);
 if(testExists && testRow[2]=="string" && PlayerPrefs.GetString(testRow[0])!=System.Text.Encoding.UTF8.GetString(System.Convert.FromBase64String(testRow[3]))) throw new System.Exception("Value mismatch "+testRow[0]);
 if(testExists && testRow[2]=="int" && PlayerPrefs.GetInt(testRow[0])!=int.Parse(testRow[3])) throw new System.Exception("Value mismatch "+testRow[0]);
}
var testGalleryField=typeof(GearCraft.Gallery.GalleryProgress).GetField("data",System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.NonPublic);
if(testGalleryField!=null) testGalleryField.SetValue(null,null);
UnityEditor.SessionState.SetBool("UXEnding20261013.active",false);
return "8 original PlayerPrefs restored and verified from file";

    }
    private static string Verify()
    {

string[] testLines=System.IO.File.ReadAllLines("/tmp/gearcraft-uxending-playerprefs-backup.txt");
var testExpected=new System.Collections.Generic.HashSet<string>(new[]{"GearCraft.Gallery.Progress.v1","GearCraft.CraftTutorial.Metrics.Craft.v5","GearCraft.CraftTutorial.Metrics.WeaponCustom.v5","GearCraft.Main.OpeningNotice.Seen.v1","GearCraft.CraftTutorial.Mastery.Craft.v5","GearCraft.CraftTutorial.Mastery.WeaponCustom.v5","GearCraft.CraftTutorial.AutoPlayed.Craft.v4","GearCraft.CraftTutorial.AutoPlayed.WeaponCustom.v3"});
var testRecords=new System.Collections.Generic.List<string[]>();
foreach(var testLine in testLines) {
 var testRow=testLine.Split('|');
 if(testRow.Length!=4 || !testExpected.Remove(testRow[0])) throw new System.Exception("Invalid backup key");
 bool.Parse(testRow[1]);
 if(testRow[2]=="string") System.Convert.FromBase64String(testRow[3]); else if(testRow[2]=="int") int.Parse(testRow[3]); else throw new System.Exception("Invalid backup type");
 testRecords.Add(testRow);
}
if(testExpected.Count!=0) throw new System.Exception("Incomplete backup");

foreach(var testRow in testRecords) {
 bool testExists=bool.Parse(testRow[1]);
 if(PlayerPrefs.HasKey(testRow[0])!=testExists) throw new System.Exception("Presence mismatch "+testRow[0]);
 if(testExists && testRow[2]=="string" && PlayerPrefs.GetString(testRow[0])!=System.Text.Encoding.UTF8.GetString(System.Convert.FromBase64String(testRow[3]))) throw new System.Exception("Value mismatch "+testRow[0]);
 if(testExists && testRow[2]=="int" && PlayerPrefs.GetInt(testRow[0])!=int.Parse(testRow[3])) throw new System.Exception("Value mismatch "+testRow[0]);
}
var testReport="8 original PlayerPrefs match after Play stop; scene="+UnityEngine.SceneManagement.SceneManager.GetActiveScene().name+" play="+UnityEditor.EditorApplication.isPlaying+" backup="+UnityEditor.SessionState.GetBool("UXEnding20261013.active",false); System.IO.File.AppendAllText("QA/UXReview20261013-Ending/observations.txt",testReport+"\n"); return testReport;
    }
}
