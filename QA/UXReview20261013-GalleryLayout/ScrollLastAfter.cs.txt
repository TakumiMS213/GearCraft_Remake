const string path="QA/UXReview20261013-GalleryLayout/scroll-after-last.txt";
var es=UnityEngine.EventSystems.EventSystem.current;
var bar=UnityEngine.Object.FindFirstObjectByType<UnityEngine.UI.Scrollbar>();
var scroll=UnityEngine.Object.FindFirstObjectByType<UnityEngine.UI.ScrollRect>();
var canvas=bar.GetComponentInParent<Canvas>();
var point=RectTransformUtility.WorldToScreenPoint(canvas.worldCamera,bar.handleRect.TransformPoint(bar.handleRect.rect.center));
var mouse=UnityEngine.InputSystem.InputSystem.AddDevice<UnityEngine.InputSystem.Mouse>("UXGalleryLayoutMouse");
var kb=UnityEngine.InputSystem.InputSystem.AddDevice<UnityEngine.InputSystem.Keyboard>("UXGalleryLayoutKeyboard");
Resources.FindObjectsOfTypeAll<UnityEditor.EditorWindow>().First(w=>w.GetType().FullName=="UnityEditor.GameView").Focus();
System.IO.File.WriteAllText(path,"Synthetic pointer click scrollbar -> keyboard Left; original list selected=Entry_24\n");
System.Action<string> log=s=>System.IO.File.AppendAllText(path,s+"\n");
System.Action<object,UnityEngine.InputSystem.InputActionChange> monitor=(o,c)=>{var a=o as UnityEngine.InputSystem.InputAction;if(a!=null&&c==UnityEngine.InputSystem.InputActionChange.ActionPerformed&&(a.name=="Click"||a.name=="Navigate"||a.name=="Move"))log("INPUT "+a.name+" control="+a.activeControl.path);};
UnityEngine.InputSystem.InputSystem.onActionChange+=monitor;
int phase=0;double start=UnityEditor.EditorApplication.timeSinceStartup;UnityEditor.EditorApplication.CallbackFunction tick=null;
System.Action finish=()=>{UnityEditor.EditorApplication.update-=tick;UnityEngine.InputSystem.InputSystem.onActionChange-=monitor;UnityEngine.InputSystem.InputSystem.RemoveDevice(mouse);UnityEngine.InputSystem.InputSystem.RemoveDevice(kb);UnityEditor.SessionState.SetBool("UXGalleryLayout20261013.scrollRunning",false);};
UnityEditor.SessionState.SetBool("UXGalleryLayout20261013.scrollRunning",true);
tick=()=>{try{double dt=UnityEditor.EditorApplication.timeSinceStartup-start;
if(dt>5){log("TIMEOUT");finish();return;}
if(phase==0&&dt>.1){UnityEngine.InputSystem.InputSystem.QueueStateEvent(mouse,new UnityEngine.InputSystem.LowLevel.MouseState{position=point});phase=1;}
else if(phase==1&&dt>.3){UnityEngine.InputSystem.InputSystem.QueueStateEvent(mouse,new UnityEngine.InputSystem.LowLevel.MouseState{position=point}.WithButton(UnityEngine.InputSystem.LowLevel.MouseButton.Left));phase=2;}
else if(phase==2&&dt>.5){UnityEngine.InputSystem.InputSystem.QueueStateEvent(mouse,new UnityEngine.InputSystem.LowLevel.MouseState{position=point});phase=3;}
else if(phase==3&&dt>.8){log("After pointer selected="+(es.currentSelectedGameObject!=null?es.currentSelectedGameObject.name:"null")+" value="+bar.value+" content="+scroll.content.anchoredPosition);ScreenCapture.CaptureScreenshot("QA/UXReview20261013-GalleryLayout/scroll-after-last-bar.png");phase=4;}
else if(phase==4&&dt>1.1){UnityEngine.InputSystem.InputSystem.QueueStateEvent(kb,new UnityEngine.InputSystem.LowLevel.KeyboardState(UnityEngine.InputSystem.Key.LeftArrow));phase=5;}
else if(phase==5&&dt>1.25){UnityEngine.InputSystem.InputSystem.QueueStateEvent(kb,new UnityEngine.InputSystem.LowLevel.KeyboardState());phase=6;}
else if(phase==6&&dt>1.6){log("After Left selected="+(es.currentSelectedGameObject!=null?es.currentSelectedGameObject.name:"null")+" value="+bar.value+" content="+scroll.content.anchoredPosition);ScreenCapture.CaptureScreenshot("QA/UXReview20261013-GalleryLayout/scroll-after-last-left.png");finish();}
}catch(System.Exception ex){log(ex.ToString());finish();}};
UnityEditor.EditorApplication.update+=tick;return "Scheduled synthetic scrollbar switch";
