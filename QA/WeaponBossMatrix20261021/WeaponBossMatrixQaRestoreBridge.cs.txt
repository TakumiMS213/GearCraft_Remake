#if UNITY_EDITOR
using System;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Security.Cryptography;
using UnityEditor;
using UnityEngine.SceneManagement;

// Temporary restoration bridge for this already-backed-up session only.
[InitializeOnLoad]
internal static class WeaponBossMatrixQaRestoreBridge
{
    private const string Folder = "QA/WeaponBossMatrix20261021/";
    private const string CommandKey = "GearCraft.QA.WeaponBossMatrix.RestoreCommand";
    private const string RestoreSha = "71894e3c5a09214eef5a724e3ae2905c1f952971e653f5eb9c2b8c8c0d6d7e22";
    private static double nextPoll;
    static WeaponBossMatrixQaRestoreBridge() { EditorApplication.update += Tick; }
    private static void Tick()
    {
        if (EditorApplication.timeSinceStartup < nextPoll || EditorApplication.isCompiling || EditorApplication.isUpdating) return;
        nextPoll = EditorApplication.timeSinceStartup + 0.5;
        string path = Folder + "local-restore-command.txt";
        if (!File.Exists(path)) return;
        try
        {
            string[] c = File.ReadAllLines(path);
            if (c.Length != 3 || c[1] != "restore") throw new InvalidOperationException("Expected id, restore, session.");
            if (SessionState.GetString(CommandKey, "") == c[0]) return;
            string session = File.ReadAllText(Folder + "session-id.txt").Trim();
            if (c[2] != session || SessionState.GetString("GearCraft.QA.WeaponBossMatrix.Session", "") != session) throw new InvalidOperationException("Wrong current session.");
            if (EditorApplication.isPlayingOrWillChangePlaymode || SceneManager.GetActiveScene().isDirty || SceneManager.GetActiveScene().name != "Title") return;
            if (!File.ReadAllText(Folder + "batch-progress.txt").StartsWith("COMPLETE")) throw new InvalidOperationException("All planned tests must finish first.");
            SessionState.SetString(CommandKey, c[0]);
            string bodyPath = Folder + "RestoreSession.cs.txt";
            using (var sha = SHA256.Create())
                if (BitConverter.ToString(sha.ComputeHash(File.ReadAllBytes(bodyPath))).Replace("-", "").ToLowerInvariant() != RestoreSha) throw new InvalidOperationException("Reviewed restoration script changed.");
            Type compiler = AppDomain.CurrentDomain.GetAssemblies().Select(a => a.GetType("MCPForUnity.Editor.Tools.ExecuteCode", false)).FirstOrDefault(t => t != null);
            if (compiler == null) throw new InvalidOperationException("Native QA compiler unavailable.");
            const BindingFlags flags = BindingFlags.Static | BindingFlags.NonPublic;
            var check = compiler.GetMethod("CheckBlockedPatterns", flags);
            var execute = compiler.GetMethod("CompileAndExecute", flags);
            if (check == null || execute == null) throw new InvalidOperationException("Native compiler checks unavailable.");
            string body = File.ReadAllText(bodyPath);
            object violation = check.Invoke(null, new object[] { body });
            if (violation != null) throw new InvalidOperationException("Native safety check: " + violation);
            object result = execute.Invoke(null, new object[] { body, "codedom" });
            if (!File.Exists(Folder + "canonical-restoration.txt")) throw new InvalidOperationException("Restoration did not verify; inspect native result: " + result);
            File.WriteAllText(Folder + "local-restore-result.txt", "VERIFIED\n" + File.ReadAllText(Folder + "canonical-restoration.txt"));
        }
        catch (Exception ex)
        {
            File.WriteAllText(Folder + "local-restore-result.txt", "FAILED\n" + ex);
        }
    }
}
#endif
