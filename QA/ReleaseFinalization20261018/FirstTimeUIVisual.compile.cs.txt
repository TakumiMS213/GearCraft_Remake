// Story and arrival messages advance on their native timers. No Camera.Render / RenderTexture.
const string folder = "QA/ReleaseFinalization20261018/";
const string session = "20261018-070128-095-8360bc7954e64d94aff83f6784baf9fb";
if (!Application.isPlaying || UnityEditor.EditorApplication.isPaused || UnityEngine.SceneManagement.SceneManager.GetActiveScene().name != "Title") throw new System.Exception("Require fresh Title Play, unpaused");

foreach (string file in System.IO.Directory.GetFiles(folder, "*progress.txt")) { string p = System.IO.File.ReadAllText(file); if (p.StartsWith("RUNNING") || p.StartsWith("LOADING")) throw new System.Exception("Another QA callback may be active: " + file); }

if (originalSelection != null && originalSelection.activeInHierarchy) events.SetSelectedGameObject(originalSelection);
UnityEngine.UI.Button hoveredButton = null;
UnityEditor.EditorApplication.CallbackFunction tick = null;
double began = UnityEditor.EditorApplication.timeSinceStartup, phaseAt = began, lastProgress = began, stableAt = began;
int stage = 0, lastFrame = -1, stableFrames = 0;
string lastScene = "", pending = "";
double captureAt = 0;
bool finished = false, modulesRestored = false, sawStory = false, sawArrival = false;
float arrowX = 0;
Color idleColor = Color.clear;
Vector3 cameraPosition = Vector3.zero;
float cameraSize = 0;
System.Action<string> note = s => evidence.AppendLine(System.DateTime.UtcNow.ToString("O") + " " + s);
System.Action<bool, string> check = (ok, text) => { note((ok ? "PASS " : "FAIL ") + text); if (!ok) throw new System.Exception(text); };
System.Func<string, UnityEngine.UI.Button> button = name => {
    var matches = UnityEngine.Object.FindObjectsByType<UnityEngine.UI.Button>(FindObjectsInactive.Exclude, FindObjectsSortMode.None).Where(b => b.name == name && b.gameObject.scene == UnityEngine.SceneManagement.SceneManager.GetActiveScene()).ToArray();
    if (matches.Length != 1) throw new System.Exception("Expected exactly one active Button: " + name + ", found " + matches.Length);
    return matches[0];
};
System.Action restoreModules = () => {
    if (modulesRestored) return;
    modulesRestored = true;
    foreach (var item in titleModules) if (item.Key != null) item.Key.enabled = item.Value;
    if (events != null) { events.sendNavigationEvents = originalNavigation; if (originalSelection != null && originalSelection.activeInHierarchy) events.SetSelectedGameObject(originalSelection); }
};
System.Action<string> finish = result => {
    if (finished) return;
    finished = true;
    if (tick != null) UnityEditor.EditorApplication.update -= tick;
    restoreModules();
    note("scene=" + UnityEngine.SceneManagement.SceneManager.GetActiveScene().name + "; play=" + Application.isPlaying + "; screen=" + Screen.width + "x" + Screen.height);
    System.IO.File.WriteAllText(output + "/result.txt", result + "\n" + evidence + "\nScope: rendered native UI / EventSystem synthetic pointer dispatch, not OS mouse or physical keyboard. Story and arrival progressed automatically. No sleep / day advance performed. Keep evidence, then stop Play and run current-session canonical restoration.\n");
    System.IO.File.WriteAllText(folder + "ui-visual-progress.txt", result + " " + run + "\n");
};
System.Action<UnityEngine.UI.Button, bool> pointer = (target, click) => {
    check(target.IsActive() && target.IsInteractable(), "button_interactable " + target.name);
    var canvas = target.GetComponentInParent<Canvas>();
    if (canvas == null) throw new System.Exception("Button has no Canvas");
    var rect = (RectTransform)target.transform;
    var eventCamera = canvas.renderMode == RenderMode.ScreenSpaceOverlay ? null : canvas.worldCamera;
    var point = RectTransformUtility.WorldToScreenPoint(eventCamera, rect.TransformPoint(rect.rect.center));
    var current = UnityEngine.EventSystems.EventSystem.current;
    var data = new UnityEngine.EventSystems.PointerEventData(current) { position = point, button = UnityEngine.EventSystems.PointerEventData.InputButton.Left, pointerId = -1, clickCount = 1, eligibleForClick = true };
    var hits = new System.Collections.Generic.List<UnityEngine.EventSystems.RaycastResult>();
    current.RaycastAll(data, hits);
    check(hits.Count > 0 && hits[0].module is UnityEngine.UI.GraphicRaycaster && hits[0].gameObject.GetComponentInParent<UnityEngine.UI.Button>() == target, "graphic_raycast_top " + target.name + " at=" + point + " hit=" + (hits.Count > 0 ? hits[0].gameObject.name : "NONE"));
    data.pointerCurrentRaycast = hits[0]; data.pointerPressRaycast = hits[0]; data.pointerEnter = target.gameObject;
    if (hoveredButton != null && hoveredButton != target) UnityEngine.EventSystems.ExecuteEvents.Execute(hoveredButton.gameObject, data, UnityEngine.EventSystems.ExecuteEvents.pointerExitHandler);
    UnityEngine.EventSystems.ExecuteEvents.Execute(target.gameObject, data, UnityEngine.EventSystems.ExecuteEvents.pointerEnterHandler);
    hoveredButton = target;
    if (!click) return;
    data.pressPosition = point; data.pointerPress = target.gameObject; data.rawPointerPress = hits[0].gameObject;
    UnityEngine.EventSystems.ExecuteEvents.Execute(target.gameObject, data, UnityEngine.EventSystems.ExecuteEvents.pointerDownHandler);
    UnityEngine.EventSystems.ExecuteEvents.Execute(target.gameObject, data, UnityEngine.EventSystems.ExecuteEvents.pointerUpHandler);
    UnityEngine.EventSystems.ExecuteEvents.Execute(target.gameObject, data, UnityEngine.EventSystems.ExecuteEvents.pointerClickHandler);
    // Button.OnPointerClick invokes its existing persistent onClick listeners exactly once.
    note("native_pointer_click " + target.name);
};
System.Action<string> capture = name => {
    pending = output + "/" + name + ".png";
    if (System.IO.File.Exists(pending)) throw new System.Exception("Refuse screenshot overwrite " + pending);
    captureAt = UnityEditor.EditorApplication.timeSinceStartup;
    ScreenCapture.CaptureScreenshot(pending, 1);
    note("capture_requested " + name + " frame=" + Time.frameCount);
};
System.Action<int> advance = value => { stage = value; phaseAt = UnityEditor.EditorApplication.timeSinceStartup; };
System.Func<RectTransform> arrow = () => {
    var target = GameObject.Find("FirstBedGuideArrow");
    return target != null ? target.GetComponent<RectTransform>() : null;
};
System.Action<RectTransform, string> logArrow = (rect, label) => {
    var canvas = rect.GetComponentInParent<Canvas>(); var cam = canvas.renderMode == RenderMode.ScreenSpaceOverlay ? null : canvas.worldCamera;
    var corners = new Vector3[4]; rect.GetWorldCorners(corners);
    var lower = RectTransformUtility.WorldToScreenPoint(cam, corners[0]); var upper = RectTransformUtility.WorldToScreenPoint(cam, corners[2]);
    check(lower.x >= 0f && lower.y >= 0f && upper.x <= Screen.width && upper.y <= Screen.height, "arrow_inside_view " + label);
    check(Mathf.Abs((lower.y + upper.y) * .5f - Screen.height * .5f) < 2f && rect.anchorMin == new Vector2(0f, .5f) && rect.anchorMax == new Vector2(0f, .5f), "arrow_left_middle_anchor " + label);
    var image = rect.GetComponent<UnityEngine.UI.Image>(); check(image != null && image.enabled && image.sprite != null && image.color.a > 0f, "arrow_has_visible_sprite " + label);
    note("arrow " + label + " anchored=" + rect.anchoredPosition + " size=" + rect.sizeDelta + " pixels=" + lower + ".." + upper);
};
note("session=" + session + "; title_selection=" + (originalSelection == null ? "NONE" : originalSelection.name));
note("Only Title InputModule and navigation temporarily disabled to avoid physical pointer overriding synthetic hover. Persistent data not directly modified; final canonical restore required.");
System.IO.File.WriteAllText(folder + "ui-visual-progress.txt", "RUNNING " + run + " stage=0\n");
tick = () => {

        if (!Application.isPlaying || UnityEditor.EditorApplication.isPaused) { finish("FAIL play_stopped_or_paused"); return; }

        } else if (stage == 2) { pointer(button("StartButton"), false); advance(3);

        } else if (stage == 6) { pointer(button("HideSetting"), true); advance(7);

            check(GameObject.Find("SettingPanel") == null, "setting_panel_closed"); restoreModules(); pointer(button("StartButton"), true); advance(8);
        } else if (stage == 8 && scene == "Story" && stable) { capture("04-opening-story"); advance(9);

            string taskText = string.Join(" / ", tasks.GetComponentsInChildren<TMPro.TMP_Text>().Select(t => t.text.Replace("\n", " ")).ToArray()); note("dorm_task_text=" + taskText);
            check(taskText.Contains("ベッドで眠って明日に備える"), "first_bed_goal_visible");
            var rect = arrow(); logArrow(rect, "a"); arrowX = rect.anchoredPosition.x; capture("05-dorm-bed-arrow-a"); advance(10);
        } else if (stage == 10 && now - phaseAt >= .3 && arrow() != null && Mathf.Abs(arrow().anchoredPosition.x - arrowX) > 1f) {
            logArrow(arrow(), "b"); note("arrow_sway_delta_x=" + (arrow().anchoredPosition.x - arrowX).ToString("F3", invariant)); capture("06-dorm-bed-arrow-b"); advance(11);

        } else if (stage == 11) { finish("PASS"); }
