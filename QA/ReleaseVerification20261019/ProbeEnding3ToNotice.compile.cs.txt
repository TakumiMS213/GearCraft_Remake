using UnityEngine;
using System.Linq;
public static class ProbeEnding3ToNoticeCompile { public static object Run() {
// C#6 execute_code body. Native Ending scene/ED3 -> native notice; stops for visual review.
const string folder = "QA/ReleaseVerification20261019/";
const string label = "ending3-native";
System.Action<bool,string> require = (ok,s) => { if (!ok) throw new System.Exception(s); };
foreach (string name in new[] { "canonical-backup-ready.txt", "typed-prefs-before.txt", "typed-prefs-before.json", "session-id.txt" })
    require(System.IO.File.Exists(folder+name), "Current session backup missing: "+name);
string session = System.IO.File.ReadAllText(folder+"session-id.txt").Trim();
var rows = System.IO.File.ReadAllLines(folder+"typed-prefs-before.txt").Where(x=>!string.IsNullOrWhiteSpace(x)).Select(x=>x.Split('|')).ToArray();
require(rows.Length==16 && rows.All(x=>x.Length==4) && rows.Select(x=>x[0]).Distinct().Count()==16, "16 typed backup keys required");
var backup = Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText(folder+"typed-prefs-before.json"));
require((string)backup["session_id"]==session && UnityEditor.SessionState.GetString("GearCraft.QA.ReleaseVerification.Session", "")==session, "Backup/session mismatch");
require(!System.IO.File.Exists(folder+"canonical-restoration.txt"), "This session has already been restored");
require(Application.isPlaying && !UnityEditor.EditorApplication.isPaused && Time.timeScale==1f && Time.captureDeltaTime==0f, "Unpaused real-time Play required");
require(UnityEngine.SceneManagement.SceneManager.GetActiveScene().name=="Title", "Begin in idle Title");
require(!EndingManager.IsLoadingEnding && EndingManager.PendingEndingNum==0, "Require fresh Play ending state");
require(StatusManager.Instance!=null, "Native Title StatusManager is required");
foreach (string file in System.IO.Directory.GetFiles(folder,"*progress.txt")) {
    string text=System.IO.File.ReadAllText(file); require(!text.StartsWith("RUNNING")&&!text.StartsWith("LOADING"), "Another QA callback: "+file);
}
foreach (string name in new[] { label+"-notice-result.txt", label+"-notice.png", label+"-restore-pending.txt", label+"-stagecounter-owned.txt" })
    require(!System.IO.File.Exists(folder+name), "Refuse evidence overwrite: "+name);
var initialTransition = UnityEngine.Object.FindFirstObjectByType<SceneTransitionManager>();
require(initialTransition!=null && !initialTransition.IsLoading, "Title transition must be idle");
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic;
var viewerType=typeof(ImageSequenceViewer);
System.Func<ImageSequenceViewer,string,object> field=(viewer,name)=>viewerType.GetField(name,flags).GetValue(viewer);
foreach(string name in new[]{"images","currentIndex","currentTween","galleryUnlockNotice","showingGalleryUnlockNotice","advanceButton","galleryNoticeButton"}) require(viewerType.GetField(name,flags)!=null,"Native field missing: "+name);
var viewType=typeof(UnityEditor.Editor).Assembly.GetType("UnityEditor.GameView");
var gameView=Resources.FindObjectsOfTypeAll(viewType).OfType<UnityEditor.EditorWindow>().FirstOrDefault();
require(gameView!=null,"Keep the GameView open for native ScreenCapture");
var sizeProperty=viewType.GetProperty("targetRenderSize",flags);
require(sizeProperty!=null,"GameView targetRenderSize unavailable");
System.Func<Vector2> renderSize=()=>gameView==null?Vector2.zero:(Vector2)sizeProperty.GetValue(gameView,null);
System.Func<string> renderMismatch=()=>{
    var size=renderSize();
    if(size.x<1||size.y<1)return "GameView targetRenderSize="+size;
    var camera=Camera.main;
    if(camera==null||Mathf.Abs(camera.pixelWidth-size.x)>.5f||Mathf.Abs(camera.pixelHeight-size.y)>.5f)return "Camera pixels="+(camera==null?"NONE":camera.pixelWidth+"x"+camera.pixelHeight)+"; GameView="+size;
    var canvases=UnityEngine.Object.FindObjectsByType<Canvas>(FindObjectsInactive.Exclude,FindObjectsSortMode.None).Where(c=>c.isRootCanvas&&c.isActiveAndEnabled&&c.targetDisplay==0&&c.renderMode!=RenderMode.WorldSpace&&c.gameObject.scene==UnityEngine.SceneManagement.SceneManager.GetActiveScene()).ToArray();
    if(canvases.Length==0)return "No active screen-space root Canvas";
    foreach(var canvas in canvases){var rect=canvas.pixelRect;if(Mathf.Abs(rect.width-size.x)>1f||Mathf.Abs(rect.height-size.y)>1f)return "Canvas "+canvas.name+" pixelRect="+rect+"; GameView="+size;}
    return "";
};
require(renderMismatch().Length==0,"GameView/Camera/Canvas mismatch: "+renderMismatch());
Vector2 capturedRenderSize=Vector2.zero;
var evidence=new System.Text.StringBuilder();
System.Action<string> note=s=>evidence.AppendLine(System.DateTime.UtcNow.ToString("O")+" "+s);
var errors=new System.Collections.Generic.List<string>();
UnityEngine.Application.LogCallback log=(message,stack,type)=>{
    string lower=message.ToLowerInvariant();
    if(type==LogType.Exception||type==LogType.Error||type==LogType.Assert||lower.Contains("dotween")&&(lower.Contains("safe mode")||lower.Contains("missing")||lower.Contains("null"))) errors.Add(type+": "+message+"\n"+stack);
};
UnityEngine.Events.UnityAction<UnityEngine.SceneManagement.Scene,UnityEngine.SceneManagement.Scene> sceneChanged=(a,b)=>note("scene "+a.name+" -> "+b.name);
UnityEditor.EditorApplication.CallbackFunction tick=null;
double wallStart=UnityEditor.EditorApplication.timeSinceStartup, lastClick=-100, captureStarted=0;
int stage=0, clicks=0, observedIndex=-99, endingHandle=0, captureFrame=0;
bool done=false;
string output=System.IO.Path.GetFullPath(folder+label+"-notice.png");
GameObject ownedCounter=null;
ImageSequenceViewer currentViewer=null;
System.Action<string> finish=outcome=>{
    if(done)return;done=true;
    if(tick!=null)UnityEditor.EditorApplication.update-=tick;
    Application.logMessageReceived-=log;
    UnityEngine.SceneManagement.SceneManager.activeSceneChanged-=sceneChanged;
    note("outcome="+outcome+"; clicks="+clicks+"; gallery_unlocked="+GearCraft.Gallery.GalleryProgress.IsUnlocked+"; errors="+errors.Count+"; callbacks_removed=true");
    foreach(string error in errors)note("ERROR "+error);
    System.IO.File.WriteAllText(folder+label+"-notice-result.txt",outcome+"\n"+evidence+"\nNative Ending scene is left at the observed state. Current session 16-key restoration remains required. Do not claim normal game completion or full UI visual PASS from this result alone.\n");
    System.IO.File.WriteAllText(folder+label+"-progress.txt",outcome+"\n");
};
System.IO.File.WriteAllText(folder+label+"-restore-pending.txt","PENDING_ROOT_RESTORE\nsession="+session+"\nNative RecordEnding may save Gallery data; root must restore fresh 16 typed keys after Play stops.\n");
try {
    // The Title does not contain StageCounter; a real Main -> Ending transition would retain one.
    // Supply only that native singleton as a QA fixture, without simulating a boss victory.
    if(StageCounter.Instance==null) {
        ownedCounter=new GameObject("__QA_Ending3NativeStageCounter");
        ownedCounter.AddComponent<StageCounter>().StageCount=31;
        System.IO.File.WriteAllText(folder+label+"-stagecounter-owned.txt",ownedCounter.GetInstanceID().ToString(System.Globalization.CultureInfo.InvariantCulture)+"\n");
        note("Fixture: native StageCounter at 31 supplied because Title lacks it; remove after Title-return verification");
    }
    note("begin session="+session+"; native_stage="+StageCounter.Instance.StageCount+"; gallery_was_unlocked="+GearCraft.Gallery.GalleryProgress.IsUnlocked);
    // Only the in-memory unlock bit is set to locked. Native EndingManager.Awake performs RecordEnding/Save.
    // Preserve all existing discovery lists, listeners, and the canonical backup.
    GearCraft.Gallery.GalleryProgress.Data.hasReachedEnding=false;
    Application.logMessageReceived+=log;
    UnityEngine.SceneManagement.SceneManager.activeSceneChanged+=sceneChanged;
    System.IO.File.WriteAllText(folder+label+"-progress.txt","RUNNING native ED3 -> notice\n");
    EndingManager.LoadEndingScene(3);
    note("invoked native EndingManager.LoadEndingScene(3), unlock fixture held only in runtime cache");
    tick=()=>{
        try {
            double now=UnityEditor.EditorApplication.timeSinceStartup;
            require(Application.isPlaying&&!UnityEditor.EditorApplication.isPaused,"Play stopped/paused externally");
            require(now-wallStart<55,"55-second wall watchdog");
            require(Time.timeScale==1f,"Unexpected timeScale");
            require(errors.Count==0,"Native error/relevant DOTween warning captured");
            string scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene().name;
            require(scene=="Title"||scene=="Ending","Unexpected scene: "+scene);
            if(stage==0) {
                if(scene!="Ending")return;
                currentViewer=UnityEngine.Object.FindFirstObjectByType<ImageSequenceViewer>();
                var manager=UnityEngine.Object.FindFirstObjectByType<EndingManager>();
                if(currentViewer==null||manager==null)return;
                var transition=UnityEngine.Object.FindFirstObjectByType<SceneTransitionManager>();
                if(transition!=null&&transition.IsLoading)return;
                var sequence=field(currentViewer,"images") as GameObject[];
                if(sequence==null||sequence.Length==0)return;
                require(EndingManager.PendingEndingNum==3&&GearCraft.Gallery.GalleryProgress.IsUnlocked,"Native ED3 did not record Gallery unlock");
                require(manager.HasPendingGalleryUnlockNotice,"First-unlock notice not pending");
                endingHandle=UnityEngine.SceneManagement.SceneManager.GetActiveScene().handle;
                note("native ED3 loaded handle="+endingHandle+"; sequence_count="+sequence.Length+"; notice_pending=true; Gallery unlocked by Awake");
                stage=1;return;
            }
            require(scene=="Ending"&&UnityEngine.SceneManagement.SceneManager.GetActiveScene().handle==endingHandle,"Ending changed before notice review");
            if(stage==1) {
                require(currentViewer!=null,"Native viewer disappeared");
                bool notice=(bool)field(currentViewer,"showingGalleryUnlockNotice");
                if(notice) {
                    var noticeObject=(GameObject)field(currentViewer,"galleryUnlockNotice");
                    var button=(UnityEngine.UI.Button)field(currentViewer,"galleryNoticeButton");
                    require(noticeObject!=null&&noticeObject.activeInHierarchy,"Notice root inactive");
                    if(button==null||!button.IsActive()||!button.IsInteractable())return;
                    var panel=noticeObject.GetComponent<UnityEngine.UI.Image>();
                    require(panel!=null&&panel.color.r<0.001f&&panel.color.g<0.001f&&panel.color.b<0.001f&&panel.color.a>0.999f,"Notice black background is not opaque black");
                    var textItems=noticeObject.GetComponentsInChildren<TMPro.TMP_Text>(false);
                    string combined=string.Join("\n",textItems.Select(x=>x.text.Replace("<br>","\n")).ToArray());
                    require(combined.Contains("ギャラリーが解放されました！")&&combined.Contains("タイトル画面から確認できます"),"Notice wording mismatch");
                    foreach(var text in textItems) { text.ForceMeshUpdate(); note("notice_text "+text.gameObject.name+"="+text.text+"; overflowing="+text.isTextOverflowing); require(!text.isTextOverflowing,"Notice text overflow"); }
                    var sequence=(GameObject[])field(currentViewer,"images");
                    require(sequence.All(x=>x==null||!x.activeInHierarchy),"Story sequence remains visible behind notice");
                    var corners=new Vector3[4]; panel.rectTransform.GetWorldCorners(corners);
                    var canvas=panel.GetComponentInParent<Canvas>();var camera=canvas.renderMode==RenderMode.ScreenSpaceOverlay?null:canvas.worldCamera;
                    var screenCorners=corners.Select(x=>RectTransformUtility.WorldToScreenPoint(camera,x)).ToArray();
                    Rect viewport=canvas.rootCanvas.pixelRect;
                    note("notice_black_rect pixel_min="+screenCorners.Min(x=>x.x)+","+screenCorners.Min(x=>x.y)+"; max="+screenCorners.Max(x=>x.x)+","+screenCorners.Max(x=>x.y)+"; canvas_viewport="+viewport+"; gameview="+renderSize()+"; canvas_order="+canvas.sortingOrder);
                    require(screenCorners.Min(x=>x.x)<=viewport.xMin+1&&screenCorners.Min(x=>x.y)<=viewport.yMin+1&&screenCorners.Max(x=>x.x)>=viewport.xMax-1&&screenCorners.Max(x=>x.y)>=viewport.yMax-1,"Black notice panel does not cover Canvas viewport");
                    require(renderMismatch().Length==0,"Capture render mismatch: "+renderMismatch());
                    capturedRenderSize=renderSize();
                    Canvas.ForceUpdateCanvases();
                    ScreenCapture.CaptureScreenshot(output,1);
                    captureStarted=now;captureFrame=Time.frameCount;stage=2;
                    note("native GameView capture requested "+output+"; frame="+captureFrame+"; notice remains displayed for review");
                    return;
                }
                var tween=field(currentViewer,"currentTween") as DG.Tweening.Tween;
                if(tween!=null&&DG.Tweening.TweenExtensions.IsActive(tween))return;
                if(now-lastClick<0.20)return;
                var advance=(UnityEngine.UI.Button)field(currentViewer,"advanceButton");
                if(advance==null||!advance.IsActive()||!advance.IsInteractable())return;
                int index=(int)field(currentViewer,"currentIndex");
                if(index!=observedIndex) {note("sequence settled index="+index+"; frame="+Time.frameCount);observedIndex=index;}
                require(UnityEngine.EventSystems.EventSystem.current!=null,"No native EventSystem");
                var pointer=new UnityEngine.EventSystems.PointerEventData(UnityEngine.EventSystems.EventSystem.current){button=UnityEngine.EventSystems.PointerEventData.InputButton.Left};
                UnityEngine.EventSystems.ExecuteEvents.Execute< UnityEngine.EventSystems.IPointerClickHandler >(advance.gameObject,pointer,UnityEngine.EventSystems.ExecuteEvents.pointerClickHandler);
                lastClick=now;clicks++;require(clicks<=20,"Unexpected sequence click count");
                return;
            }
            if(stage==2) {
                require(now-captureStarted<12,"ScreenCapture file did not complete");
                if(Time.frameCount<=captureFrame||!System.IO.File.Exists(output))return;
                byte[] png=System.IO.File.ReadAllBytes(output);
                if(png.Length<33)return;
                require(png[0]==137&&png[1]==80&&png[2]==78&&png[3]==71,"Capture is not PNG");
                int w=(png[16]<<24)|(png[17]<<16)|(png[18]<<8)|png[19];int h=(png[20]<<24)|(png[21]<<16)|(png[22]<<8)|png[23];
                require(w==Mathf.RoundToInt(capturedRenderSize.x)&&h==Mathf.RoundToInt(capturedRenderSize.y),"Capture dimensions differ from requested GameView targetRenderSize");
                note("capture completed "+w+"x"+h+"; bytes="+png.Length+"; manual visual review still required");
                finish("READY_NOTICE_REVIEW");
            }
        } catch(System.Exception ex) { note(ex.ToString());finish("FAIL"); }
    };
    UnityEditor.EditorApplication.update+=tick;
} catch(System.Exception ex) {note(ex.ToString());finish("FAIL");}
return "Scheduled native ED3 sequence and notice capture. Will hold at notice for image review; then run ProbeEndingNoticeToTitle.cs.txt. Current session restoration remains required.";

} }
