// Read-only per-game-frame observations. No transform/physics synchronization or bot changes.
string geometryPath=folder+label+"-geometry.csv",projectilePath=folder+label+"-projectiles-first-seen.csv";
var geometryBuffer=new System.Text.StringBuilder();
var projectileSeen=new System.Collections.Generic.HashSet<int>();
var geometryTerrain=UnityEngine.Object.FindObjectsByType<Collider2D>(FindObjectsSortMode.None).Where(c=>c!=null&&!c.isTrigger&&c.CompareTag("Ground")).ToArray();
var geometryGate=GateManager.Instance;
var geometryGateCollider=geometryGate!=null?geometryGate.GetComponent<Collider2D>():null;
float geometryLastFlush=Time.time;
System.IO.File.WriteAllText(geometryPath,"game_seconds,frame,stage,hp,gate,attack_events,player_x,player_y,velocity_x,velocity_y,player_min_x,player_min_y,player_max_x,player_max_y,player_floor_top,player_floor_depth,boss_id,boss_name,boss_hp,boss_x,boss_y,boss_min_x,boss_min_y,boss_max_x,boss_max_y,boss_floor_top,boss_floor_depth,player_boss_overlap_x,player_boss_overlap_y,gate_min_x,gate_max_x,gate_min_y,gate_max_y,previous_attack_suppression\n");
System.IO.File.WriteAllText(projectilePath,"game_seconds,frame,stage,observation,projectile_id,name,damage,position_x,position_y,velocity_x,velocity_y,min_x,min_y,max_x,max_y,gate_min_x,gate_max_x,gate_min_y,gate_max_y,gate_overlap_x,gate_overlap_y,horizontal_gap_to_gate,boss_id,boss_min_x,boss_min_y,boss_max_x,boss_max_y\n");
System.Func<Bounds,float> geometryFloor=b=>{
 float top=float.NegativeInfinity;
 foreach(var c in geometryTerrain)if(c!=null&&c.enabled&&c.gameObject.activeInHierarchy){var t=c.bounds;if(t.min.x<=b.max.x&&t.max.x>=b.min.x)top=Mathf.Max(top,t.max.y);}
 return top;
};
System.Action geometryFlush=()=>{if(geometryBuffer.Length>0){System.IO.File.AppendAllText(geometryPath,geometryBuffer.ToString());geometryBuffer.Length=0;}geometryLastFlush=Time.time;};
System.Action<EnemyController[]> observeGeometry=enemies=>{
 if(player==null||playerCollider==null)return;
 var bosses=enemies.Where(e=>e!=null&&e.HP>0f&&e.enemyData!=null&&e.enemyData.IsBossType).ToArray();
 if(bosses.Length==0)return;
 Bounds p=playerCollider.bounds,g=geometryGateCollider!=null?geometryGateCollider.bounds:default(Bounds);
 float floor=geometryFloor(p);
 foreach(var e in bosses){
  var ec=e.GetComponent<Collider2D>();if(ec==null)continue;Bounds b=ec.bounds;float bf=geometryFloor(b);
  var values=new[]{number(Time.time-started),Time.frameCount.ToString(),(StageCounter.Instance!=null?StageCounter.Instance.StageCount:lastStage).ToString(),number(status.HP),number(status.GATE),attackEvents.ToString(),number(player.transform.position.x),number(player.transform.position.y),number(body.linearVelocity.x),number(body.linearVelocity.y),number(p.min.x),number(p.min.y),number(p.max.x),number(p.max.y),number(floor),number(floor-p.min.y),e.GetInstanceID().ToString(),e.enemyData.enemyName,number(e.HP),number(e.transform.position.x),number(e.transform.position.y),number(b.min.x),number(b.min.y),number(b.max.x),number(b.max.y),number(bf),number(bf-b.min.y),number(Mathf.Max(0f,Mathf.Min(p.max.x,b.max.x)-Mathf.Max(p.min.x,b.min.x))),number(Mathf.Max(0f,Mathf.Min(p.max.y,b.max.y)-Mathf.Max(p.min.y,b.min.y))),number(g.min.x),number(g.max.x),number(g.min.y),number(g.max.y),attackSuppression};
  geometryBuffer.AppendLine(string.Join(",",values.Select(csv).ToArray()));
 }
 var bb=bosses.FirstOrDefault(e=>e.enemyData.enemyName=="BB-413");
 if(bb!=null){
  var allowed=new System.Collections.Generic.HashSet<string>((bb.enemyData.bulletPrefabs??new GameObject[0]).Where(go=>go!=null).Select(go=>go.name));
  Bounds b=bb.GetComponent<Collider2D>().bounds;
  foreach(var projectile in UnityEngine.Object.FindObjectsByType<EnemyBulletController>(FindObjectsSortMode.None)){
   if(projectile==null||!projectile.gameObject.activeInHierarchy||!allowed.Contains(projectile.name.Replace("(Clone)",""))||!projectileSeen.Add(projectile.GetInstanceID()))continue;
   var c=projectile.GetComponent<Collider2D>();if(c==null)continue;var rb=projectile.GetComponent<Rigidbody2D>();Bounds q=c.bounds;Vector2 v=rb!=null?rb.linearVelocity:Vector2.zero;
   var values=new[]{number(Time.time-started),Time.frameCount.ToString(),(StageCounter.Instance!=null?StageCounter.Instance.StageCount:lastStage).ToString(),"first_editor_frame_not_instant_spawn",projectile.GetInstanceID().ToString(),projectile.name,number(projectile.bulletDamage),number(projectile.transform.position.x),number(projectile.transform.position.y),number(v.x),number(v.y),number(q.min.x),number(q.min.y),number(q.max.x),number(q.max.y),number(g.min.x),number(g.max.x),number(g.min.y),number(g.max.y),number(Mathf.Max(0f,Mathf.Min(q.max.x,g.max.x)-Mathf.Max(q.min.x,g.min.x))),number(Mathf.Max(0f,Mathf.Min(q.max.y,g.max.y)-Mathf.Max(q.min.y,g.min.y))),number(q.min.x-g.max.x),bb.GetInstanceID().ToString(),number(b.min.x),number(b.min.y),number(b.max.x),number(b.max.y)};
   System.IO.File.AppendAllText(projectilePath,string.Join(",",values.Select(csv).ToArray())+"\n");
  }
 }
 if(Time.time-geometryLastFlush>=1f)geometryFlush();
};
